while (cur != null || stack.length) {
  while (cur != null) {          // ← 一路向左
    stack.push(cur);              //    每走一步就存档
    cur = cur.left;
  }
  cur = stack.pop();              // ← 弹出（回到上一层）
  visit(cur);                     // ← 访问节点
  cur = cur.right;                // ← 转向右子树
}
